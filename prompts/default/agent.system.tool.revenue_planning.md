evaluate monetization ideas before acting
use for opportunity screening and lane triage
reject anything with weak legality consent provenance or platform fit

~~~json
{
    "thoughts": [
        "Need to screen this lane before executing it."
    ],
    "tool_name": "revenue_planning",
    "tool_args": {
        "opportunity_name": "Inbox-to-CRM assistant for opt-in leads",
        "legality": "high",
        "consent": "high",
        "data_provenance": "high",
        "tos_alignment": "high",
        "time_to_cash": "medium",
        "expected_margin": "medium",
        "repeatability": "high",
        "automation_fit": "high",
        "defensibility": "medium",
        "notes": "First-party inbound messages only."

}
~~~
